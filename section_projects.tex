    % 2. GPT-2 From Scratch
	\project
	{GPT-2 Language Model - From-Scratch Implementation}{2026}
	{\website{https://github.com/krstff/transformer}{GitHub}}
	{Built and trained a complete GPT-2 transformer (83M parameters) entirely from scratch in PyTorch.
	\begin{itemize}
		\item Implemented all core components: multi-head masked self-attention, pre-LayerNorm transformer blocks, feed-forward layers with GELU activation, token and positional embeddings.
    \item Pre-trained on 3.2 billion tokens (OpenWebText subset) reaching a stable cross-entropy loss of ~3.75; fine-tuned for haiku style mimicry.
    \item Mixed-precision (bfloat16) training loop with gradient clipping, automated loss logging, and multi-GPU support via \texttt{nn.DataParallel}.
	\end{itemize}
	}
	{PyTorch, Transformers, Attention-mechanism, GPU Optimization}
